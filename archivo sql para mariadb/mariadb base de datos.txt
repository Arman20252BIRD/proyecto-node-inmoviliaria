-- Base de datos del Sistema Inmobiliario (compatible con el backend Node.js)
-- Nombre de BD debe coincidir con DB_NAME del archivo .env
CREATE DATABASE IF NOT EXISTS inmobiliaria_db CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE inmobiliaria_db;

-- Eliminar tablas existentes para evitar conflictos de llaves foráneas
DROP TABLE IF EXISTS autenticacion;
DROP TABLE IF EXISTS fotografias;
DROP TABLE IF EXISTS publicaciones;
DROP TABLE IF EXISTS propiedades;
DROP TABLE IF EXISTS usuarios;

-- 1. Tabla de Usuarios (Con teléfono y foto de perfil según el documento de la UTM)
CREATE TABLE usuarios (
  id INT(11) NOT NULL AUTO_INCREMENT,
  nombre_completo VARCHAR(150) NOT NULL,
  correo VARCHAR(100) NOT NULL,
  telefono VARCHAR(20) DEFAULT NULL,
  contrasena VARCHAR(255) NOT NULL,
  fotografia_perfil VARCHAR(255) DEFAULT NULL,
  rol ENUM('usuario', 'administrador') NOT NULL DEFAULT 'usuario',
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY correo (correo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 2. Tabla de Autenticación (Facebook OAuth, WebAuthn y 2FA reconocimiento facial)
CREATE TABLE autenticacion (
  id INT(11) NOT NULL AUTO_INCREMENT,
  usuario_id INT(11) NOT NULL,
  metodo VARCHAR(50) NOT NULL,
  identificador_externo VARCHAR(255) DEFAULT NULL,
  datos_secretos TEXT DEFAULT NULL,
  PRIMARY KEY (id),
  KEY usuario_id (usuario_id),
  CONSTRAINT autenticacion_ibfk_1 FOREIGN KEY (usuario_id) REFERENCES usuarios (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 3. Tabla de Propiedades (Con superficie, habitaciones, baños y contacto exigidos por la rúbrica)
CREATE TABLE propiedades (
  id INT(11) NOT NULL AUTO_INCREMENT,
  usuario_id INT(11) NOT NULL,
  titulo VARCHAR(150) NOT NULL,
  descripcion TEXT NOT NULL,
  ubicacion VARCHAR(255) NOT NULL,
  precio DECIMAL(12,2) NOT NULL,
  superficie_terreno DECIMAL(10,2) DEFAULT NULL,
  superficie_construccion DECIMAL(10,2) DEFAULT NULL,
  habitaciones INT(11) DEFAULT NULL,
  banos INT(11) DEFAULT NULL,
  contacto_vendedor VARCHAR(100) NOT NULL,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY usuario_id (usuario_id),
  CONSTRAINT propiedades_ibfk_1 FOREIGN KEY (usuario_id) REFERENCES usuarios (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 4. Tabla de Fotografías (Mínimo tres fotografías por propiedad)
CREATE TABLE fotografias (
  id INT(11) NOT NULL AUTO_INCREMENT,
  propiedad_id INT(11) NOT NULL,
  url_imagen VARCHAR(255) NOT NULL,
  PRIMARY KEY (id),
  KEY propiedad_id (propiedad_id),
  CONSTRAINT fotografias_ibfk_1 FOREIGN KEY (propiedad_id) REFERENCES propiedades (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 5. Tabla de Publicaciones (Control de aprobación del administrador: pendiente, aprobada, rechazada)
CREATE TABLE publicaciones (
  id INT(11) NOT NULL AUTO_INCREMENT,
  propiedad_id INT(11) NOT NULL,
  estado ENUM('pendiente', 'aprobada', 'rechazada') NOT NULL DEFAULT 'pendiente',
  observaciones TEXT DEFAULT NULL,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY propiedad_id (propiedad_id),
  CONSTRAINT publicaciones_ibfk_1 FOREIGN KEY (propiedad_id) REFERENCES propiedades (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
